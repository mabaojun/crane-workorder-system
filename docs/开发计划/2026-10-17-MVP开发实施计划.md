# 济南诚重吊装业务管理系统 — MVP 开发实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: 按本计划逐任务执行。步骤使用 checkbox（`- [ ]`）语法跟踪进度。每个任务对应一次独立 Trae 会话，完成后验收通过再进入下一个任务。

**Goal:** 按 MVP 分级完成济南诚重吊装业务管理系统核心闭环：工程骨架 → 认证账号 → 档案 → 需求单 → 工单状态机 → 派单接单 → 打卡工时 → 异常上报 → 后台管理前端 → 司机/客户移动端，达到「需求单转工单、派单、接单、执行、结单核心链路闭环」的验收标准（见文档 10 §6）。

**Architecture:** Spring Boot 3 分层单体（`backend/` 按业务域分包），MySQL 8 单库，前端 pnpm monorepo（`frontend/` 下 `apps/admin` + `apps/screen` + `apps/mobile` + `packages/shared`）。互斥约束与冲突提醒为二期任务（T-17/T-18/T-32/T-34），本期不实现。契约以文档 01/02/03 为唯一基准，视觉以文档 11 为唯一基准。

**Tech Stack:** Java 17 + Spring Boot 3.x + MyBatis-Plus + MySQL 8；Vue 3 + Vite + Element Plus + Pinia（后台）；uni-app Vue3 语法（移动端）；Vite + ECharts（大屏骨架）；pnpm 工作区。

**计划输入文档（执行前必读）:** `docs/前置设计/` 下 00（规划/就绪标准）、01（HLD/契约基准）、02（数据库）、03（接口契约）、08（部署）、09（任务包规范）、10（MVP 分级）、11（前端设计手册）。各任务引用具体章节。

## 0. Trae 会话执行规则（每个任务必须遵守）

来源：文档 09 §1 使用纪律 + Trae 会话式开发特点。

- **一次会话一个任务**：禁止合并任务；本计划中每个「任务」= 一次独立会话
- **契约只读**：不得修改 01 §6.3 枚举、03 接口路径与错误码；发现缺口先改文档再改代码
- **目录约束**：后端产出落在 `backend/src/main/java/com/crane/`，前端落在 `frontend/`，禁止越界
- **视觉约束**：前端不得使用 Emoji、外部 CDN/字体/图片；图标用共享层内联 SVG sprite；令牌以 11 号手册为准
- **命名约束**：表与字段 snake_case、类 PascalCase、方法/文件 kebab-case；提交 `type(scope): subject`
- **提交粒度**：每任务完成并验收后提交一次（或按逻辑分组提交），禁止跨任务混杂
- **启动验证**：后端用 `mvn spring-boot:run` 或 IDE；前端用 `pnpm dev`；每次验收必须实际运行命令

```java
package com.crane.common.result;

import lombok.AllArgsConstructor;
import lombok.Data;
import lombok.NoArgsConstructor;
import java.util.UUID;

@Data
@NoArgsConstructor
@AllArgsConstructor
public class ApiResult<T> {
    private int code;
    private String message;
    private T data;
    private String traceId;

    public static <T> ApiResult<T> ok(T data) {
        return new ApiResult<>(0, "OK", data, UUID.randomUUID().toString().substring(0, 8));
    }

    public static ApiResult<Void> ok() {
        return new ApiResult<>(0, "OK", null, UUID.randomUUID().toString().substring(0, 8));
    }

    public static <T> ApiResult<T> fail(int code, String message) {
        return new ApiResult<>(code, message, null, UUID.randomUUID().toString().substring(0, 8));
    }
}
```

```java
// backend/src/main/java/com/crane/common/exception/GlobalExceptionHandler.java
package com.crane.common.exception;

import com.crane.common.result.ApiResult;
import lombok.extern.slf4j.Slf4j;
import org.springframework.validation.FieldError;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;

@Slf4j
@RestControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(BizException.class)
    public ApiResult<Void> handleBiz(BizException e) {
        return ApiResult.fail(e.getCode(), e.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ApiResult<Void> handleValid(MethodArgumentNotValidException e) {
        FieldError fe = e.getBindingResult().getFieldError();
        String msg = fe == null ? "参数校验失败" : fe.getField() + ": " + fe.getDefaultMessage();
        return ApiResult.fail(ErrorCode.PARAM_INVALID.getCode(), msg);
    }

    @ExceptionHandler(Exception.class)
    public ApiResult<Void> handleOther(Exception e) {
        log.error("系统异常", e);
        return ApiResult.fail(ErrorCode.SYSTEM_ERROR.getCode(), "系统内部错误");
    }
}
```

```java
    @Value("${crane.jwt.secret:crane-secret-key-please-change-in-prod-0123456789}")
    private String secret;

    @Value("${crane.jwt.expire-hours:720}")
    private long expireHours;

    private SecretKey key() {
        return Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    public String createToken(Long userId, String role) {
        Date now = new Date();
        Date expiry = new Date(now.getTime() + expireHours * 3600_000L);
        return Jwts.builder()
                .subject(String.valueOf(userId))
                .claim("role", role)
                .issuedAt(now)
                .expiration(expiry)
                .signWith(key())
                .compact();
    }

    public Claims parse(String token) {
        return Jwts.parser().verifyWith(key()).build().parseSignedClaims(token).getPayload();
    }
}
```

```java
package com.crane.common.auth;

import com.crane.common.exception.BizException;
import com.crane.common.exception.ErrorCode;
import io.jsonwebtoken.Claims;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Component;
import org.springframework.web.servlet.HandlerInterceptor;

@Component
@RequiredArgsConstructor
public class AuthInterceptor implements HandlerInterceptor {

    private final JwtUtil jwtUtil;

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        String header = request.getHeader("Authorization");
        if (header == null || !header.startsWith("Bearer ")) {
            throw new BizException(ErrorCode.UNAUTHORIZED);
        }
        try {
            Claims claims = jwtUtil.parse(header.substring(7));
            request.setAttribute("userId", Long.valueOf(claims.getSubject()));
            request.setAttribute("role", claims.get("role", String.class));
            return true;
        } catch (Exception e) {
            throw new BizException(ErrorCode.TOKEN_EXPIRED);
        }
    }
}
```

```java
// backend/src/main/java/com/crane/modules/auth/service/AuthService.java
package com.crane.modules.auth.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.crane.common.auth.JwtUtil;
import com.crane.common.exception.BizException;
import com.crane.common.exception.ErrorCode;
import com.crane.modules.auth.dto.LoginRequest;
import com.crane.modules.auth.dto.LoginResponse;
import com.crane.modules.auth.entity.SysUser;
import com.crane.modules.auth.mapper.SysUserMapper;
import lombok.RequiredArgsConstructor;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.stereotype.Service;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class AuthService {

    private final SysUserMapper sysUserMapper;
    private final JwtUtil jwtUtil;
    private final BCryptPasswordEncoder encoder = new BCryptPasswordEncoder();

    public LoginResponse passwordLogin(LoginRequest req) {
        SysUser user = sysUserMapper.selectOne(new LambdaQueryWrapper<SysUser>()
                .eq(SysUser::getPhone, req.getPhone()));
        if (user == null || !encoder.matches(req.getPassword(), user.getPassword())) {
            throw new BizException(10012, "手机号或密码错误");
        }
        if (user.getStatus() == null || user.getStatus() != 1) {
            throw new BizException(10013, "账号已停用");
        }
        String token = jwtUtil.createToken(user.getId(), user.getRoleCode());
        return LoginResponse.builder()
                .token(token).userId(user.getId())
                .roleCode(user.getRoleCode()).nickname(user.getNickname())
                .build();
    }

    public SysUser me(Long userId) {
        SysUser user = sysUserMapper.selectById(userId);
        if (user == null) {
            throw new BizException(ErrorCode.NOT_FOUND);
        }
        user.setPassword(null);
        return user;
    }

    public void logout(Long userId) {
        // 更新强制下线时间戳，使已签发令牌失效（03 API-AUTH-005 说明）
        SysUser user = new SysUser();
        user.setId(userId);
        user.setForceLogoutAt(LocalDateTime.now());
        sysUserMapper.updateById(user);
    }
}
```

```java
import org.springframework.transaction.annotation.Transactional;

import java.util.Set;

@Service
@RequiredArgsConstructor
public class StaffService {

    private static final Set<String> VEHICLE_STATUS = Set.of("IDLE", "IN_PROGRESS", "MAINTENANCE", "DISABLED", "THIRD_PARTY");

    private final StaffMapper staffMapper;
    private final VehicleMapper vehicleMapper;
    private final VehicleBindingMapper bindingMapper;

    public Staff createStaff(Staff staff) {
        staff.setId(null);
        staffMapper.insert(staff);
        return staff;
    }

    public Staff updateStaff(Staff staff) {
        staffMapper.updateById(staff);
        return staffMapper.selectById(staff.getId());
    }

    public void deleteStaff(Long id) {
        staffMapper.deleteById(id);
    }

    public Vehicle createVehicle(Vehicle vehicle) {
        if (!VEHICLE_STATUS.contains(vehicle.getStatus())) {
            throw new BizException(ErrorCode.PARAM_INVALID);
        }
        vehicle.setId(null);
        vehicleMapper.insert(vehicle);
        return vehicle;
    }

    public Vehicle updateVehicle(Vehicle vehicle) {
        if (vehicle.getStatus() != null && !VEHICLE_STATUS.contains(vehicle.getStatus())) {
            throw new BizException(ErrorCode.PARAM_INVALID);
        }
        vehicleMapper.updateById(vehicle);
        return vehicleMapper.selectById(vehicle.getId());
    }

    @Transactional
    public VehicleBinding bind(Long vehicleId, Long staffId) {
        // 同一车辆同一时段仅一个生效绑定（02 §3.3.15 跨行约束，服务层校验）
        Long active = bindingMapper.selectCount(new LambdaQueryWrapper<VehicleBinding>()
                .eq(VehicleBinding::getVehicleId, vehicleId)
                .eq(VehicleBinding::getIsActive, 1));
        if (active != null && active > 0) {
            throw new BizException(10020, "车辆当前已有生效绑定");
        }
        VehicleBinding binding = new VehicleBinding();
        binding.setVehicleId(vehicleId);
        binding.setStaffId(staffId);
        binding.setIsActive(1);
        bindingMapper.insert(binding);
        return binding;
    }
}
```

```java
// backend/src/main/java/com/crane/modules/workorder/service/WorkOrderStateMachine.java
package com.crane.modules.workorder.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.crane.common.exception.BizException;
import com.crane.common.exception.ErrorCode;
import com.crane.modules.workorder.entity.WorkOrder;
import com.crane.modules.workorder.entity.WorkOrderStatusLog;
import com.crane.modules.workorder.mapper.WorkOrderMapper;
import com.crane.modules.workorder.mapper.WorkOrderStatusLogMapper;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.util.Map;
import java.util.Set;

@Service
@RequiredArgsConstructor
public class WorkOrderStateMachine {

    // 合法流转：key = from_status，value = 允许的 to_status 集合
    private static final Map<String, Set<String>> TRANSITIONS = Map.of(
            "PENDING_ACCEPT", Set.of("PENDING_ARRIVE", "CANCELLED"),
            "PENDING_ARRIVE", Set.of("IN_PROGRESS", "CANCELLED"),
            "IN_PROGRESS",    Set.of("SUSPENDED", "COMPLETED"),
            "SUSPENDED",      Set.of("IN_PROGRESS", "CANCELLED"),
            "COMPLETED",      Set.of(),
            "CANCELLED",      Set.of()
    );

    private final WorkOrderMapper workOrderMapper;
    private final WorkOrderStatusLogMapper statusLogMapper;

    @Transactional
    public WorkOrder transition(WorkOrder order, String targetStatus, Long operatorId, String remark) {
        String from = order.getStatus();
        if (!TRANSITIONS.getOrDefault(from, Set.of()).contains(targetStatus)) {
            throw new BizException(10030, "非法状态流转: " + from + " -> " + targetStatus);
        }
        // 互斥校验：进入 IN_PROGRESS 时，同一司机不能已有另一个 IN_PROGRESS 工单
        if ("IN_PROGRESS".equals(targetStatus) && order.getStaffId() != null) {
            Long active = workOrderMapper.selectCount(new LambdaQueryWrapper<WorkOrder>()
                    .eq(WorkOrder::getStaffId, order.getStaffId())
                    .eq(WorkOrder::getStatus, "IN_PROGRESS")
                    .ne(WorkOrder::getId, order.getId()));
            if (active != null && active > 0) {
                throw new BizException(ErrorCode.CONFLICT);
            }
        }
        order.setStatus(targetStatus);
        workOrderMapper.updateById(order);

        WorkOrderStatusLog log = new WorkOrderStatusLog();
        log.setOrderId(order.getId());
        log.setFromStatus(from);
        log.setToStatus(targetStatus);
        log.setOperatorId(operatorId);
        log.setRemark(remark);
        statusLogMapper.insert(log);
        return order;
    }
}
```

```java
    private final VehicleMapper vehicleMapper;
    private final StaffMapper staffMapper;
    private final WorkOrderMapper workOrderMapper;

    public List<Map<String, Object>> precheck(Long vehicleId, Long staffId) {
        List<Map<String, Object>> checks = new ArrayList<>();
        // VEHICLE_IDLE 车辆空闲
        Vehicle vehicle = vehicleId == null ? null : vehicleMapper.selectById(vehicleId);
        checks.add(Map.of("code", "VEHICLE_IDLE", "pass",
                vehicle != null && "IDLE".equals(vehicle.getStatus())));
        // STAFF_ON_DUTY 司机在岗
        Staff staff = staffId == null ? null : staffMapper.selectById(staffId);
        checks.add(Map.of("code", "STAFF_ON_DUTY", "pass",
                staff != null && staff.getStatus() != null && staff.getStatus() == 1));
        // STAFF_NO_ACTIVE_ORDER 司机无进行中工单（互斥口径）
        Long active = workOrderMapper.selectCount(new LambdaQueryWrapper<WorkOrder>()
                .eq(WorkOrder::getStaffId, staffId)
                .eq(WorkOrder::getStatus, "IN_PROGRESS"));
        checks.add(Map.of("code", "STAFF_NO_ACTIVE_ORDER", "pass", active == null || active == 0));
        // TIME_VALID / PENDING_DISPATCH_LIMIT 按 03 API-DISP-001 补充
        return checks;
    }

    public boolean allPass(List<Map<String, Object>> checks) {
        return checks.stream().allMatch(c -> Boolean.TRUE.equals(c.get("pass")));
    }
}
```

- [ ] **Step 2: 打卡服务（签到距离校验 ≤ 签到范围阈值；阈值存配置表二期，本期常量 2000 米）**

```java
// backend/src/main/java/com/crane/modules/attendance/service/AttendanceService.java
package com.crane.modules.attendance.service;

import com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper;
import com.crane.common.exception.BizException;
import com.crane.modules.attendance.entity.Attendance;
import com.crane.modules.attendance.mapper.AttendanceMapper;
import com.crane.modules.workorder.entity.WorkOrder;
import com.crane.modules.workorder.mapper.WorkOrderMapper;
import com.crane.modules.workorder.service.WorkOrderStateMachine;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

@Service
@RequiredArgsConstructor
public class AttendanceService {

    private static final double MAX_DISTANCE_M = 2000.0; // 签到范围阈值（二期接入配置表）

    private final AttendanceMapper attendanceMapper;
    private final WorkOrderMapper workOrderMapper;
    private final WorkOrderStateMachine stateMachine;

    public Attendance checkIn(Long orderId, Long staffId, Double lng, Double lat) {
        WorkOrder order = workOrderMapper.selectById(orderId);
        if (order == null || !"PENDING_ARRIVE".equals(order.getStatus())) {
            throw new BizException(10040, "工单状态不允许签到");
        }
        double distance = haversineM(lng, lat, order.getLongitude(), order.getLatitude());
        if (distance > MAX_DISTANCE_M) {
            throw new BizException(10041, "距离施工地点过远，无法签到");
        }
        Attendance att = new Attendance();
        att.setOrderId(orderId);
        att.setStaffId(staffId);
        att.setType("CHECK_IN");
        att.setLongitude(lng);
        att.setLatitude(lat);
        att.setDistanceM(distance);
        attendanceMapper.insert(att);

        // 签到成功 → 工单进入进行中
        stateMachine.transition(order, "IN_PROGRESS", staffId, "司机签到");
        return att;
    }

    private double haversineM(double lng1, double lat1, double lng2, double lat2) {
        if (lng2 == null || lat2 == null) {
            return 0;
        }
        double R = 6371000;
        double dLat = Math.toRadians(lat2 - lat1);
        double dLng = Math.toRadians(lng2 - lng1);
        double a = Math.sin(dLat / 2) * Math.sin(dLat / 2)
                + Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2))
                * Math.sin(dLng / 2) * Math.sin(dLng / 2);
        return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }
}
```

```ts
export const useAuthStore = defineStore('auth', () => {

    token.value = ''
    localStorage.removeItem('crane_token')
  }

  return { token, role, login, logout }
```
